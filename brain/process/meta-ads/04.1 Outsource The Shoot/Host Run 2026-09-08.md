## Appendix — the composite (local ffmpeg)

```python
"""Local composite for WP_P3_HOST_triedthemall_9x16_v1 (480p test).
Inputs: clip_1.mp4, clip_2.mp4 (concat → creator_full.mp4), screenrec.mp4 (5.2 s), words.json, disclosure.png."""
import json, re, subprocess, os
D=os.path.dirname(os.path.abspath(__file__)); W,H=1080,1920
WBOX=int(W*0.78); HBOX=int(H*0.50); CY=int(H*0.47); DISC_Y=290
dur=lambda f: float(subprocess.check_output(["ffprobe","-v","error","-show_entries","format=duration","-of","default=nw=1:nk=1",os.path.join(D,f)]))
t1=dur("clip_1.mp4")
words=json.load(open(os.path.join(D,"words.json")))
norm=lambda t: re.sub(r"[^a-z0-9']","",t.lower())
t_closer=next(s for s,e,w in words if s>=t1-0.5 and norm(w).startswith("authentic"))
s_in=t1+0.3; e_in=t_closer-0.3
fps=subprocess.check_output(["ffprobe","-v","error","-select_streams","v:0","-show_entries","stream=r_frame_rate","-of","default=nw=1:nk=1",os.path.join(D,"creator_full.mp4")]).decode().strip()
fc=(f"[0:v]scale={W}:{H}:flags=lanczos[base];"
    f"[1:v]crop=iw:ih*0.95:0:ih*0.05,scale={WBOX}:{HBOX}:force_original_aspect_ratio=decrease,tpad=stop_mode=clone:stop_duration=10,setpts=PTS+{s_in}/TB[rec];"
    f"[base][rec]overlay=(W-w)/2:{CY}-h/2:enable='between(t,{s_in:.3f},{e_in:.3f})':eof_action=pass[v1];"
    f"[v1][2:v]overlay=0:{DISC_Y}[vout]")
subprocess.check_call(["ffmpeg","-y","-v","error","-i",os.path.join(D,"creator_full.mp4"),"-i",os.path.join(D,"screenrec.mp4"),"-i",os.path.join(D,"disclosure.png"),
     "-filter_complex",fc,"-map","[vout]","-map","0:a","-c:v","libx264","-crf","18","-preset","medium","-pix_fmt","yuv420p","-profile:v","high","-r",fps,"-c:a","copy",os.path.join(D,"final.mp4")])
```
