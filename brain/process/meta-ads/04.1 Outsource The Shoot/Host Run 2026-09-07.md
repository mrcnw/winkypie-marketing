Three things that bit, and their fixes:

1. **Preset nudge.** `generate_video` refused the first submission with a "looks like the
   preset IN THE DARK" notice. Resubmit with `declined_preset_id`; the models stay locked.
2. **Whisper hears the brand as "Winky Pied"** and the caption QA gate (`group_captions.py`,
   exit 3) blocks the burn. Merge the two words into `WinkyPie` in `words.json` before
   grouping — in the script below. Never `--allow-qa-misses` for this: it would burn "PIED".
3. **The bundle's `make_captions.py` has no `--no-caps`** (the subtitles reference is ahead of

   default, which sits exactly on the disclosure pill for the whole hook — replace `\N` with a

### Composite — local, ffmpeg

Card crop boxes on the raw 1206 × 2622 screenshots: check `03-photo-check-pass.png` →
`(0, 1000, 1206, 2608)`; pose `02-pose-mimic-add-your-photo.png` → `(0, 330, 1206, 1938)`;
result `05-result-pose-and-photo-used.png` → `(0, 330, 1206, 2200)` plus the §11.2 band.

```python
"""Local composite for WP_P2_HOST_freecheck_9x16_v1.
Inputs (same dir): creator_full.mp4, words.json ([[start,end,word],...]),
cards/card_check_pass.png, cards/card_pose.png, cards/card_result_disclosed.png, disclosure.png
Output: final.mp4 (one encode, audio copied)."""
import json, re, subprocess, sys, os
D=os.path.dirname(os.path.abspath(__file__)); C=os.path.join(os.path.dirname(D),"cards")
W,H=1080,1920
WBOX=int(W*0.78); HBOX=int(H*0.50); CY=int(H*0.47)      # card box + vertical centre
DISC_Y=290                                                 # disclosure strip top (just under Meta's top 14%)
words=json.load(open(os.path.join(D,"words.json")))
norm=lambda t: re.sub(r"[^a-z0-9']","",t.lower())
def find(tok, after=0.0):
    for s,e,w in words:
        if s>=after and norm(w).startswith(tok): return s
    raise SystemExit(f"word not found: {tok} after {after}")
T=float(subprocess.check_output(["ffprobe","-v","error","-show_entries","format=duration","-of","default=nw=1:nk=1",os.path.join(D,"creator_full.mp4")]))
fps=subprocess.check_output(["ffprobe","-v","error","-select_streams","v:0","-show_entries","stream=r_frame_rate","-of","default=nw=1:nk=1",os.path.join(D,"creator_full.mp4")]).decode().strip()
t_winky=find("winky"); t_if=find("if",t_winky); t_rest=find("rest",t_if); t_pick=find("pick",t_rest)
t_one=find("one",t_pick); t_three=find("three",t_one)
cards=[
 ("card_check_pass.png", t_winky, t_if-0.3),                              # hero: WinkyPie ... it's a go
 ("card_pose.png",       t_pick,  min(t_pick+1.5, t_one-0.3)),
 ("card_result_disclosed.png", t_one, max(t_one+2.6, t_three-0.2)),     # hold >=2.5 s, through the brand line
]
for n,s,e in cards: print(f"{n}: {s:.2f} -> {e:.2f} ({e-s:.2f}s)")
inputs=["-i",os.path.join(D,"creator_full.mp4")]
fc=[]; prev="[0:v]"
for i,(n,s,e) in enumerate(cards,1):
    inputs+=["-i",os.path.join(C,n)]
    fc.append(f"[{i}:v]scale={WBOX}:{HBOX}:force_original_aspect_ratio=decrease[c{i}]")
    fc.append(f"{prev}[c{i}]overlay=(W-w)/2:{CY}-h/2:enable='between(t,{s:.3f},{e:.3f})'[v{i}]"); prev=f"[v{i}]"
inputs+=["-i",os.path.join(D,"disclosure.png")]
k=len(cards)+1
fc.append(f"{prev}[{k}:v]overlay=0:{DISC_Y}[vout]")
cmd=["ffmpeg","-y",*inputs,"-filter_complex",";".join(fc),"-map","[vout]","-map","0:a","-c:v","libx264","-crf","18","-preset","medium","-pix_fmt","yuv420p","-profile:v","high","-r",fps,"-c:a","copy",os.path.join(D,"final.mp4")]
print(" ".join(cmd)); subprocess.check_call(cmd)
print("T=%.2f fps=%s -> final.mp4"%(T,fps))
```

```bash
# Runs INSIDE the Higgsfield sandbox. Placeholders: __FINAL_URL__ (hosted final.mp4), __PUT_URL__ (presigned PUT for final_captioned.mp4)
set -e; mkdir -p output && cd output
curl -sL -o final.mp4 '__FINAL_URL__'
cat > script.txt <<'TXT'
Would your selfie pass? Most AI photo apps charge you for the bad result. WinkyPie checks the selfie first. Free. Green tick, it's a go. If it won't work, it says why. Retake. Nothing charged. The rest is quick. Pick a pose, it tells you how to stand. One selfie. Pro photos. No photographer. 3 days free. Cancel anytime.
TXT
printf 'Would your selfie pass?' > hook.txt
python3 $HF_WORKFLOWS/ugc-website-video/scripts/transcribe_words.py final.mp4 -o words.json
python3 - <<'PY'
import json, re, difflib
norm = lambda t: re.sub(r"[^a-z0-9]", "", t.lower())
words = json.load(open("words.json"))
# Whisper hears the brand as "Winky Pied" — merge it back into one word or the caption QA gate blocks
merged = []; i = 0
while i < len(words):
    if i+1 < len(words) and norm(words[i][2]) == "winky" and norm(words[i+1][2]).startswith("pie"):
        merged.append([words[i][0], words[i+1][1], "WinkyPie"]); i += 2
    else:
        merged.append(words[i]); i += 1
words = merged; json.dump(words, open("words.json","w"))
hook = open("hook.txt").read().split()
a, b = [norm(w[2]) for w in words], [norm(t) for t in hook]
i_end = 0
for tag, i1, i2, j1, j2 in difflib.SequenceMatcher(a=a, b=b, autojunk=False).get_opcodes():
    if tag in ("equal", "replace"): i_end = max(i_end, i2)
first_body = i_end if 0 < i_end < len(words) else min(len(hook), len(words) - 1)
plate_end = round(words[first_body][0], 2); hook_start = round(words[0][0], 2)
json.dump({"video": "final.mp4", "segments": [{"start": hook_start, "end": plate_end, "text": open("hook.txt").read().strip()}]}, open("hook_seg.json", "w"))
json.dump([w for w in words if w[0] >= plate_end - 1e-6], open("body_words.json", "w"))
print(f"plate {hook_start}-{plate_end}s; body starts at {plate_end}s; words={len(words)}")
PY
python3 $HF_WORKFLOWS/ugc-website-video/scripts/group_captions.py body_words.json -o segments.json --video final.mp4 --script script.txt
python3 $HF_WORKFLOWS/ugc-website-video/scripts/make_captions.py segments.json -o captions.ass --size 50
python3 $HF_WORKFLOWS/ugc-website-video/scripts/make_captions.py hook_seg.json -o hook.ass --size 56
python3 - <<'PY'
H = 1920
def fix(p, align, margin, one_line=False):
    out = []
    for ln in open(p):
        ln = ln.rstrip("\n")
        if ln.startswith("Style: Cap,"):
            parts = ln.split(","); parts[-5] = str(align); parts[-2] = str(margin); ln = ",".join(parts)
        if one_line and ln.startswith("Dialogue:"):
            ln = ln.replace("\\N", " ")
        out.append(ln)
    open(p, "w").write("\n".join(out) + "\n")
fix("captions.ass", 2, round(0.22 * H))   # bottom captions, raised: bottom edge ~0.78H
fix("hook.ass", 8, round(0.09 * H), one_line=True)   # top hook plate, ONE line
PY
ffmpeg -y -v error -i final.mp4 -vf "ass=captions.ass,ass=hook.ass" -c:v libx264 -crf 18 -preset medium -pix_fmt yuv420p -profile:v high -level 4.0 -c:a copy final_captioned.mp4
ls -la final_captioned.mp4 && cat words.json | head -c 400; echo
```
